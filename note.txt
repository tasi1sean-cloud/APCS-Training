#include <bits/stdc++.h>

typedef long long ll;
char m1;

int ans=0;
vector<vector<pair<int,int>>> vt;
void bfs(int x,int y){
  queue<pair<int,int>> q;
  q.push({x,y});
  while(!q.empty()){
    int nx=q.front().first,ny=q.front().second;
    int now=ny;
    q.pop();
    int s=vt[nx].size();
    for (int i=0;i<s;i++){
      q.push({vt[nx][i].first,now+vt[nx][i].second});
    }
    ans=max(ans,now);
  }
}

vector<vector<int>> ans;
    vector<int> vt;
    
    void dfs(int start, int n, int k){
        if(vt.size() == k){
            ans.push_back(vt);
            return ;
        }
         for(int i = start; i <= n; ++i){
            vt.push_back(i);
            dfs(i+1, n, k);
            vt.pop_back();
        }
    }

    vector<vector<int>> combine(int n, int k) {
        dfs(1, n, k);
        return ans;
